function burnOf(act, minutes, intensity, kg) {
  const base = ACTS.find(a => a.id === act);
  if (!base) return 0;
  const f = (INTENSITY.find(i => i[0] === intensity) || INTENSITY[1])[2];
  return Math.max(0, Math.round((((base.met * f) - 1) * 3.5 * (kg || 70) / 200) * minutes));
}

function computeTargets(p, burn = 0) {
  const w = +p.weight || 70, ht = +p.height || 170, age = +p.age || 30;
  const bmr = 10 * w + 6.25 * ht - 5 * age + (p.sex === "male" ? 5 : -161);
  const added = p.burnMode === "added";
  const act = added ? 1.2 : ({ low: 1.2, light: 1.375, mid: 1.55, high: 1.725 }[p.activity] || 1.375);
  const tdee = bmr * act;
  let kcal = tdee;
  if (p.goal === "lose") kcal = tdee - (p.rate === "steady" ? 550 : 275);
  if (p.goal === "gain") kcal = tdee + (p.rate === "steady" ? 400 : 250);
  const baseKcal = Math.round(kcal / 10) * 10;
  const extra = added ? Math.round(burn) : 0;
  const floor = p.sex === "male" ? 1500 : 1300;
  const finalKcal = Math.max(floor, baseKcal + extra);
  const protein = Math.round(w * (p.goal === "maintain" ? 1.6 : 1.8));
  const fat = Math.round((finalKcal * 0.27) / 9);
  const carbs = Math.max(40, Math.round((finalKcal - protein * 4 - fat * 9) / 4));
  return { kcal: finalKcal, base: Math.max(floor, baseKcal), extra, protein, carbs, fat,
    tdee: Math.round(tdee), floored: finalKcal === floor, added };
}

function shrink(file, max, q) {
  return new Promise((resolve, reject) => {
    const fr = new FileReader();
    fr.onerror = () => reject(new Error("read"));
    fr.onload = () => {
      const img = new Image();
      img.onerror = () => reject(new Error("decode"));
      img.onload = () => {
        const s = Math.min(1, max / Math.max(img.width, img.height));
        const c = document.createElement("canvas");
        c.width = Math.round(img.width * s); c.height = Math.round(img.height * s);
        c.getContext("2d").drawImage(img, 0, 0, c.width, c.height);
        resolve(c.toDataURL("image/jpeg", q));
      };
      img.src = fr.result;
    };
    fr.readAsDataURL(file);
  });
}

{"is_food":true,"dish":"שם מסודר למנה","portion":"הכמות שהנחת","confidence":"high"|"medium"|"low","items":[{"name":"מרכיב","grams":0,"kcal":0,"protein":0,"carbs":0,"fat":0}],"note":"מה לא היה ברור, משפט אחד"}

אם זה לא תיאור של אוכל, החזר {"is_food":false,"note":"הסבר קצר"}.` }]
  });
  const txt = (data.content || []).filter(b => b.type === "text").map(b => b.text).join("");
  const clean = txt.replace(/```json|```/g, "").trim();
  const a = clean.indexOf("{"), b = clean.lastIndexOf("}");
  if (a < 0) throw new Error("parse");
  const r = JSON.parse(clean.slice(a, b + 1));
  if (r.is_food === false) return r;
  const items = (r.items || []).map(i => ({ ...i }));
  const sum = items.reduce((acc, i) => ({
    kcal: acc.kcal + (+i.kcal || 0), protein: acc.protein + (+i.protein || 0),
    carbs: acc.carbs + (+i.carbs || 0), fat: acc.fat + (+i.fat || 0)
  }), { kcal: 0, protein: 0, carbs: 0, fat: 0 });
  if (!sum.kcal) throw new Error("parse");
  return { ...r, items, sum };
}

async function askAdvice(history, ctx) {
  const data = await callAPI({
    system: adviceSystem(ctx),

  });
  const blocks = (data.content || []).filter(b => b.type === "text");

  const seen = {}, sources = [];
  blocks.forEach(b => (b.citations || []).forEach(c => {
    const u = c.url || c.source;

function slotsFor(rem) {
  const out = [];
  if (rem.meals) rem.mealTimes.forEach((t, i) => out.push({ key: "m" + i, time: t, title: "זמן ארוחה",
    body: ["בוקר טוב! מה בצלחת?", "הפסקת צהריים — אל תדלגו", "ארוחת ערב. צלמו ונרשום"][i] || "זמן ארוחה" }));
  if (rem.water) {
    const [fh, fm] = rem.waterFrom.split(":").map(Number);
    const [th] = rem.waterTo.split(":").map(Number);
    for (let h = fh; h <= th; h += Math.max(1, rem.waterEvery))
      out.push({ key: "w" + h, time: `${String(h).padStart(2, "0")}:${String(fm).padStart(2, "0")}`,
        title: "כוס מים", body: "רגע של מים. הגוף אומר תודה" });
  }
  return out.sort((a, b) => a.time < b.time ? -1 : 1);
}

function EatItApp() {
  const [ready, setReady] = useState(false);
  const [accounts, setAccounts] = useState([]);
  const [active, setActive] = useState(null);

  useEffect(() => {
    let alive = true;
    const bail = setTimeout(() => { if (alive) setReady(true); }, 3000);
    (async () => {
      try {
        const [accs, act] = await Promise.all([store.get("eatit:accounts"), store.get("eatit:active")]);
        if (!alive) return;
        const list = Array.isArray(accs) ? accs : [];
        setAccounts(list);
        if (act && list.some(a => a.id === act)) setActive(act);
      } catch { /* fresh */ }
      if (alive) { clearTimeout(bail); setReady(true); }
    })();
    return () => { alive = false; clearTimeout(bail); };
  }, []);

  const createAccount = (name, avatar, seed) => {
    const acc = { id: uid(), name: name.trim(), avatar, created: Date.now() };
    const list = [...accounts, acc];
    setAccounts(list); store.set("eatit:accounts", list);
    if (seed) DATA_KEYS.forEach(k => { if (seed[k] != null) store.set(kx(acc.id, k), seed[k]); });
    setActive(acc.id); store.set("eatit:active", acc.id);
  };
  const signIn = id => { setActive(id); store.set("eatit:active", id); };

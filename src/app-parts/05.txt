/* ------------------------------ diary ------------------------------ */
function Diary({ date, setDate, meals, totals, targets, workouts, burn, onRemoveMove,

  const strip = useMemo(() => Array.from({ length: 7 }, (_, i) => addDays(new Date(), i - 6)), []);
  const pct = totals.kcal / targets.kcal;
  const left = targets.kcal - totals.kcal;
  const isToday = keyOf(date) === keyOf(new Date());
  const headline =
    totals.kcal === 0 ? "יום חדש, צלחת נקייה" :
    pct < 0.55 ? "מתקדמים יפה" :
    pct < 0.95 ? "כמעט שם, נשאר מקום לעוד" :
    pct <= 1.08 ? "בול על היעד. כל הכבוד!" : "יום מלא — מחר ממשיכים";

  return (
    <>
      <div className="datestrip">
        {strip.map(d => (
          <button key={keyOf(d)} className="daybtn" data-on={keyOf(d) === keyOf(date) ? 1 : 0} onClick={() => setDate(d)}>
            <small>{DAYS_HE[d.getDay()]}</small><b>{d.getDate()}</b>
          </button>
        ))}
      </div>

      <div className="panel tilt-a" style={{ marginBottom: 14, padding: "16px 14px" }}>
        <div style={{ display: "flex", alignItems: "center", gap: 6 }}>
          <div style={{ flex: 1 }}>
            <div className="tag">{left >= 0 ? "נשארו להיום" : "מעל היעד ב"}</div>
            <div style={{ display: "flex", alignItems: "baseline", gap: 7 }}>
              <span className="big" style={{ color: left >= 0 ? "var(--chili)" : "var(--gold)" }}>{Math.abs(r0(left))}</span>
              <span style={{ fontSize: 13.5, color: "var(--muted)" }}>מתוך {targets.kcal}</span>
            </div>
            {targets.extra > 0 && (
              <div className="chip" style={{ marginTop: 4, display: "inline-block", color: "var(--sage)" }}>
                כולל {targets.extra} מהאימון
              </div>

          ))}

      <div className="panel tilt-b" style={{ marginBottom: 14 }}>

          <span className="hand" style={{ fontSize: 22 }}>מים היום</span>

        <button className="btn btn-ghost btn-sm" onClick={onAddMove}><Plus size={16} /> אימון</button>
      </div>
      {workouts.length === 0
        ? <div className="panel" style={{ padding: "12px 14px", marginBottom: 14, fontSize: 13.5, color: "var(--muted)" }}>

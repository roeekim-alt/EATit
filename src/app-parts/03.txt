  const signOut = () => { setActive(null); store.set("eatit:active", null); };
  const removeAccount = id => {
    const list = accounts.filter(a => a.id !== id);
    setAccounts(list); store.set("eatit:accounts", list);
    DATA_KEYS.forEach(k => store.set(kx(id, k), null));
    if (active === id) signOut();
  };
  const renameAccount = (id, name, avatar) => {
    const list = accounts.map(a => a.id === id ? { ...a, name, avatar } : a);
    setAccounts(list); store.set("eatit:accounts", list);
  };

  if (!ready) return (
    <div className="eatit" style={{ display: "flex", alignItems: "center", justifyContent: "center", minHeight: "100vh" }}>
      <style>{CSS}</style>
      <div style={{ textAlign: "center" }} className="bob">
        <Bowl pct={0.5} size={110} /><div className="mark" style={{ marginTop: 6 }}>eatit</div>
      </div>
    </div>
  );

  if (!active) return (
    <div className="eatit"><style>{CSS}</style>
      <SignIn accounts={accounts} onCreate={createAccount} onSignIn={signIn} onRemove={removeAccount} />
    </div>
  );

  const me = accounts.find(a => a.id === active);
  return <Session key={active} me={me} onSignOut={signOut} onRemove={removeAccount} onRename={renameAccount} />;
}

/* ---------------------------- session ------------------------------ */
function Session({ me, onSignOut, onRemove, onRename }) {
  const [ready, setReady] = useState(false);
  const [profile, setProfile] = useState(null);
  const [log, setLog] = useState({});
  const [moves, setMoves] = useState({});
  const [water, setWater] = useState({});
  const [weights, setWeights] = useState({});
  const [chat, setChat] = useState([]);
  const [rem, setRem] = useState(DEFAULT_REM);
  const [date, setDate] = useState(() => new Date());
  const [tab, setTab] = useState("diary");
  const [sheet, setSheet] = useState(null);
  const fired = useRef({});

  useEffect(() => {
    let alive = true;
    const bail = setTimeout(() => { if (alive) setReady(true); }, 3000);
    (async () => {
      try {
        const vals = await Promise.all(DATA_KEYS.map(k => store.get(kx(me.id, k))));
        if (!alive) return;
        const [p, l, mv, wt, wg, rm, ch] = vals;
        if (p && p.weight) setProfile(p);
        if (l && typeof l === "object") setLog(l);
        if (mv && typeof mv === "object") setMoves(mv);
        if (wt && typeof wt === "object") setWater(wt);
        if (wg && typeof wg === "object") setWeights(wg);
        if (rm && typeof rm === "object") setRem({ ...DEFAULT_REM, ...rm });
        if (Array.isArray(ch)) setChat(ch);
      } catch { /* fresh */ }
      if (alive) { clearTimeout(bail); setReady(true); }
    })();
    return () => { alive = false; clearTimeout(bail); };
  }, [me.id]);

  useEffect(() => {
    if (typeof window === "undefined" || !("Notification" in window)) return;
    const tick = () => {
      if (Notification.permission !== "granted") return;
      const now = new Date();
      const hhmm = `${String(now.getHours()).padStart(2, "0")}:${String(now.getMinutes()).padStart(2, "0")}`;
      slotsFor(rem).forEach(s => {
        const id = keyOf(now) + s.key;
        if (s.time === hhmm && !fired.current[id]) {
          fired.current[id] = 1;
          try { new Notification(s.title, { body: s.body }); } catch { /* blocked */ }
        }
      });
    };
    const iv = setInterval(tick, 25000);
    return () => clearInterval(iv);
  }, [rem]);

  const put = (setter, name) => v => { setter(v); store.set(kx(me.id, name), v); };
  const saveProfile = put(setProfile, "profile");
  const saveLog = put(setLog, "log");
  const saveMoves = put(setMoves, "moves");
  const saveWater = put(setWater, "water");
  const saveWeights = put(setWeights, "weights");
  const saveRem = put(setRem, "reminders");
  const saveChat = put(setChat, "chat");

  const dk = keyOf(date);
  const meals = useMemo(() => log[dk] || [], [log, dk]);
  const workouts = useMemo(() => moves[dk] || [], [moves, dk]);
  const burn = useMemo(() => workouts.reduce((a, w) => a + w.kcal, 0), [workouts]);
  const targets = useMemo(() => profile ? computeTargets(profile, burn) : null, [profile, burn]);
  const totals = useMemo(() => meals.reduce((a, m) => ({
    kcal: a.kcal + m.kcal, protein: a.protein + m.protein, carbs: a.carbs + m.carbs, fat: a.fat + m.fat
  }), { kcal: 0, protein: 0, carbs: 0, fat: 0 }), [meals]);

  /* what the user eats most, for one-tap repeats */
  const recent = useMemo(() => {
    const map = {};
    Object.keys(log).sort().slice(-30).forEach(k => (log[k] || []).forEach(m => {
      const key = (m.dish || "").trim();
      if (!key) return;
      if (!map[key]) map[key] = { dish: key, kcal: m.kcal, protein: m.protein, carbs: m.carbs, fat: m.fat, photo: m.photo, count: 0 };
      map[key].count++;
      if (m.photo) map[key].photo = m.photo;
    }));
    return Object.values(map).sort((a, b) => b.count - a.count).slice(0, 6);
  }, [log]);

  /* one context string, shared by the suggester and the advice chat */
  const ctx = useMemo(() => profile && targets ? [
    `שם: ${me.name}`,
    `${profile.sex === "male" ? "זכר" : "נקבה"}, גיל ${profile.age}, ${profile.height} ס״מ, ${profile.weight} ק״ג`,
    `מטרה: ${{ lose: "ירידה במשקל", maintain: "שמירה", gain: "עלייה במסה" }[profile.goal]}, קצב ${profile.rate === "steady" ? "יציב" : "רגוע"}`,
    profile.prefs ? `העדפות והגבלות אכילה: ${profile.prefs}` : "",
    `יעד יומי: ${targets.kcal} קק״ל · חלבון ${targets.protein}g · פחמימות ${targets.carbs}g · שומן ${targets.fat}g`,
    `נאכלו היום: ${r0(totals.kcal)} קק״ל (חלבון ${r0(totals.protein)}g, פחמימות ${r0(totals.carbs)}g, שומן ${r0(totals.fat)}g)`,
    `נותרו: ${r0(targets.kcal - totals.kcal)} קק״ל · חלבון ${Math.max(0, targets.protein - r0(totals.protein))}g · פחמימות ${Math.max(0, targets.carbs - r0(totals.carbs))}g · שומן ${Math.max(0, targets.fat - r0(totals.fat))}g`,
    `השעה כעת: ${new Date().getHours()}:00`,
    meals.length ? `ארוחות היום: ${meals.map(m => `${m.dish} (${r0(m.kcal)})`).join(", ")}` : "עוד לא נרשמו ארוחות היום",
    workouts.length ? `אימונים היום: ${workouts.map(w => `${w.name} ${w.minutes} דק׳ (${w.kcal} קק״ל)`).join(", ")}` : "לא נרשם אימון היום",
    recent.length ? `מנות שהוא אוכל הרבה: ${recent.map(m => m.dish).join(", ")}` : ""
  ].filter(Boolean).join("\n") : "", [me, profile, targets, totals, meals, workouts, recent]);

  const prune = obj => {
    const cutoff = keyOf(addDays(new Date(), -60));
    Object.keys(obj).forEach(k => { if (k < cutoff) delete obj[k]; });
    return obj;
  };
  const addMeal = m => saveLog(prune({ ...log, [dk]: [...(log[dk] || []), m] }));
  const updateMeal = (id, patch) => saveLog({ ...log, [dk]: (log[dk] || []).map(m => m.id === id ? { ...m, ...patch } : m) });
  const removeMeal = id => saveLog({ ...log, [dk]: (log[dk] || []).filter(m => m.id !== id) });
  const addMove = w => saveMoves(prune({ ...moves, [dk]: [...(moves[dk] || []), w] }));
  const removeMove = id => saveMoves({ ...moves, [dk]: (moves[dk] || []).filter(w => w.id !== id) });

  if (!ready) return (
    <div className="eatit" style={{ display: "flex", alignItems: "center", justifyContent: "center", minHeight: "100vh" }}>
      <style>{CSS}</style><div className="bob"><Bowl pct={0.5} size={110} /></div>
    </div>
  );

  if (!profile) return (
    <div className="eatit"><style>{CSS}</style><Onboarding me={me} onDone={saveProfile} /></div>
  );

  return (
    <div className="eatit">
      <style>{CSS}</style>
      <div className="wrap" style={tab === "chat" ? { paddingBottom: 104 } : undefined}>
        <div className="topbar">
          <div style={{ display: "flex", alignItems: "center", gap: 9 }}>
            <Mascot kind={me.avatar} size={38} />
            <div>
              <div className="mark" style={{ fontSize: 23 }}>eatit</div>
              <div className="tag" style={{ fontSize: 15 }}>שלום, {me.name}</div>
            </div>
          </div>
          <button className="iconbtn" onClick={() => setTab("goals")} aria-label="יעדים והגדרות"><Settings2 size={19} /></button>
        </div>

        <div className="tabs">
          {[["diary", "היום"], ["chat", "ייעוץ"], ["trend", "מגמה"], ["goals", "יעדים"]].map(([id, t]) => (
            <button key={id} className="tab" data-on={tab === id ? 1 : 0} onClick={() => setTab(id)}>{t}</button>
          ))}
        </div>

        {tab === "diary" && (
          <Diary date={date} setDate={setDate} meals={meals} totals={totals} targets={targets}
            workouts={workouts} burn={burn} onRemoveMove={removeMove}
            cups={water[dk] || 0} cupGoal={waterGoal(profile.weight)} onCups={n => saveWater({ ...water, [dk]: n })}
            onEdit={m => setSheet({ kind: "edit", meal: m })} onDelete={removeMeal}
            onAddMove={() => setSheet({ kind: "move" })}
            onAdd={addMeal} recent={recent} ctx={ctx} prefs={profile.prefs} />
        )}
        {tab === "chat" && (
          <Advice me={me} chat={chat} onChat={saveChat} ctx={ctx} />
        )}
        {tab === "trend" && (
          <Trend log={log} moves={moves} profile={profile} weights={weights}
            onWeight={kg => saveWeights({ ...weights, [keyOf(new Date())]: kg })} />

/* ----------------------------- account ----------------------------- */
function Account({ me, onSignOut, onRename, onRemoveAcc, bundle }) {
  const [name, setName] = useState(me.name);
  const [avatar, setAvatar] = useState(me.avatar);
  const [code, setCode] = useState("");
  const [copied, setCopied] = useState(false);
  const dirty = name.trim() !== me.name || avatar !== me.avatar;

  function makeCode() {
    try {
      setCode(btoa(unescape(encodeURIComponent(JSON.stringify({ name: me.name, avatar: me.avatar, data: bundle.data })))));
    } catch { setCode(""); }
  }
  function copy() {
    try { navigator.clipboard.writeText(code); setCopied(true); setTimeout(() => setCopied(false), 1800); } catch { /* manual copy */ }
  }

  return (
    <div className="panel tilt-b" style={{ marginTop: 14 }}>
      <div style={{ display: "flex", alignItems: "center", gap: 8, marginBottom: 12 }}>
        <span className="numdot">5</span><span className="hand" style={{ fontSize: 23 }}>הפרופיל שלי</span>
      </div>

      <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 12 }}>
        <Mascot kind={avatar} size={52} />
        <div style={{ flex: 1 }}>
          <label className="label" htmlFor="acc-rename">שם</label>
          <input id="acc-rename" className="field" value={name} onChange={e => setName(e.target.value)} />
        </div>
      </div>
      <div style={{ display: "grid", gridTemplateColumns: "repeat(4,1fr)", gap: 8, marginBottom: 10 }}>
        {AVATARS.map(a => (
          <button key={a.id} className="avatarbtn" data-on={avatar === a.id ? 1 : 0} onClick={() => setAvatar(a.id)}>
            <Mascot kind={a.id} size={34} /><span style={{ fontSize: 11.5 }}>{a.t}</span>
          </button>
        ))}
      </div>
      <button className="btn btn-ghost btn-sm" style={{ width: "100%", marginBottom: 14 }} disabled={!dirty}
        onClick={() => onRename(me.id, name.trim(), avatar)}>שמרו שינויים בפרופיל</button>

      <div className="setrow" style={{ display: "block" }}>
        <div className="tag" style={{ marginBottom: 6 }}>קוד גיבוי</div>
        <p style={{ fontSize: 12.5, lineHeight: 1.6, color: "var(--muted)", margin: "0 0 8px" }}>
          הנתונים נשמרים על המכשיר הזה. קוד הגיבוי מאפשר לשחזר את הפרופיל במכשיר אחר.
        </p>
        {!code
          ? <button className="btn btn-ghost btn-sm" style={{ width: "100%" }} onClick={makeCode}>
              <Link2 size={16} /> צרו קוד גיבוי
            </button>
          : <>
              <div className="code">{code}</div>
              <button className="btn btn-ghost btn-sm" style={{ width: "100%", marginTop: 8 }} onClick={copy}>
                {copied ? <><Check size={16} /> הועתק</> : <><Copy size={16} /> העתיקו את הקוד</>}
              </button>
            </>}
      </div>

      <div style={{ display: "flex", gap: 8, marginTop: 12 }}>
        <button className="btn btn-ghost btn-sm" style={{ flex: 1 }} onClick={onSignOut}>
          <LogOut size={16} /> החלפת פרופיל
        </button>
        <button className="btn btn-ghost btn-sm" style={{ color: "var(--chili)" }}
          onClick={() => window.confirm("למחוק את הפרופיל וכל הנתונים שלו?") && onRemoveAcc()}>
          <Trash2 size={16} /> מחיקה
        </button>
      </div>
    </div>
  );
}

function Reminders({ rem, onRem, cupGoal }) {

  const [perm, setPerm] = useState(supported ? Notification.permission : "unsupported");
  const set = (k, v) => onRem({ ...rem, [k]: v });
  const setMealTime = (i, v) => onRem({ ...rem, mealTimes: rem.mealTimes.map((t, j) => j === i ? v : t) });
  const count = slotsFor(rem).length;

    if (!supported) return;

  return (
    <div className="panel tilt-b" style={{ marginBottom: 14 }}>

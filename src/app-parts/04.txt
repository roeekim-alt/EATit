/* ------------------------------ advice ----------------------------- */
const STARTERS = [
  "כמה חלבון אני צריך ביום ולמה?",
  "מה כדאי לאכול אחרי אימון כוח?",
  "יש לי עוד קלוריות פנויות — מה מומלץ?",
  "כמה קלוריות באמת שורפים בהליכה של שעה?"
];

function Advice({ me, chat, onChat, ctx }) {
  const [text, setText] = useState("");
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState("");
  const endRef = useRef(null);

  useEffect(() => { endRef.current?.scrollIntoView?.({ behavior: "smooth", block: "end" }); }, [chat.length, busy]);

  async function send(q) {
    const body = (q ?? text).trim();
    if (!body || busy) return;
    setErr(""); setText("");
    const next = [...chat, { role: "user", text: body }];
    onChat(next); setBusy(true);
    try {
      const r = await askAdvice(next, ctx);
      onChat([...next, { role: "assistant", text: r.text, sources: r.sources }].slice(-40));
    } catch (e) {
      setErr(e.message === "net"
        ? "הייעוץ לא הגיב. בדקו חיבור לאינטרנט ונסו שוב."
        : "משהו השתבש בתשובה. נסו לנסח מחדש.");
    } finally { setBusy(false); }
  }

  return (
    <>
      <div className="panel tilt-a sage" style={{ marginBottom: 14, display: "flex", gap: 11, alignItems: "center" }}>
        <MessageCircle size={22} style={{ color: "var(--sage)", flex: "0 0 auto" }} />
        <div>
          <div className="hand" style={{ fontSize: 22 }}>ייעוץ תזונה, מסביב לשעון</div>
          <p style={{ fontSize: 12.5, lineHeight: 1.55, margin: "2px 0 0", color: "var(--muted)" }}>
            רואה את היעדים והיומן שלכם. מבסס מספרים על USDA, משרד הבריאות, ACSM ו‑ISSN — ומצרף קישור למקור.
          </p>
        </div>
      </div>

      {chat.length === 0 && (
        <div style={{ marginBottom: 12 }}>
          <div className="tag" style={{ marginBottom: 8 }}>אפשר להתחיל מ־</div>
          {STARTERS.map(q => (
            <button key={q} className="qchip" onClick={() => send(q)}>{q}</button>
          ))}
        </div>
      )}

      {chat.map((m, i) => (
        <div key={i} className={"bub " + (m.role === "user" ? "bub-u" : "bub-a")}>
          {m.text}
          {m.sources?.length > 0 && (
            <div style={{ marginTop: 7, borderTop: "1.5px dashed var(--edge)", paddingTop: 6 }}>
              {m.sources.map(s => (
                <a key={s.url} className="src" href={s.url} target="_blank" rel="noopener noreferrer">
                  <ExternalLink size={11} /> {s.title}
                </a>
              ))}
            </div>
          )}
        </div>
      ))}

      {busy && (
        <div className="bub bub-a" style={{ display: "flex", alignItems: "center", gap: 9 }}>
          <Mascot kind={me.avatar} size={26} />
          <span className="dots"><span /><span /><span /></span>
        </div>
      )}

      {err && (
        <div className="panel" style={{ borderColor: "var(--chili)", display: "flex", gap: 9, marginBottom: 10 }}>
          <AlertCircle size={18} style={{ color: "var(--chili)", flex: "0 0 auto", marginTop: 2 }} />
          <span style={{ fontSize: 13.5, lineHeight: 1.5 }}>{err}</span>
        </div>
      )}

      {chat.length > 0 && (
        <p style={{ fontSize: 11.5, lineHeight: 1.6, color: "var(--muted)", margin: "6px 4px 0" }}>
          ייעוץ כללי, לא תחליף לדיאטן/ית או רופא/ה. במצב רפואי, הריון או תרופות — כדאי להתייעץ אישית.
        </p>
      )}
      <div ref={endRef} />

      <div className="composer">
        <div className="composer-in">
          <input className="field" value={text} onChange={e => setText(e.target.value)}
            onKeyDown={e => { if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); send(); } }}
            placeholder="שאלו כל דבר על תזונה ואימונים" aria-label="שאלה ליועץ" />
          <button className="btn btn-sage" style={{ minWidth: 54, padding: "0 14px" }}
            disabled={busy || !text.trim()} onClick={() => send()} aria-label="שליחה">
            {busy ? <Loader2 size={19} className="spin" /> : <Send size={19} />}
          </button>
        </div>
      </div>
    </>
  );
}

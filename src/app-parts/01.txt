function Bowl({ pct, size = 132 }) {

  const h = Math.min(p, 1.12) * 30;
  const full = p >= 0.98, over = p > 1.12;
  const seeds = [[38, 44], [55, 38], [72, 39], [88, 45], [46, 41], [64, 35], [80, 41]];
  return (
    <svg viewBox="0 0 120 104" width={size} height={size * 104 / 120} aria-hidden="true">
      {p > 0.02 && (<>
        <path d={`M20,52 Q60,${52 - h * 1.9} 100,52 Z`} fill="var(--gold)" stroke="var(--ink)" strokeWidth="2.2" strokeLinejoin="round" />
        <path d={`M28,50 Q60,${52 - h * 1.35} 92,50`} fill="none" stroke="#B9812A" strokeWidth="1.6" strokeLinecap="round" opacity=".65" />
        {p > 0.32 && seeds.slice(0, Math.round(p * 6)).map(([x, y], i) => (
          <circle key={i} cx={x} cy={52 - h * (1.9 - Math.abs(x - 60) / 42 * 1.6) * (y / 52) + 4} r="1.9"
            fill={i % 2 ? "var(--sage)" : "var(--chili)"} />
        ))}
      </>)}
      <ellipse cx="60" cy="52" rx="41" ry="7.5" fill="var(--panel)" stroke="var(--ink)" strokeWidth="2.4" />
      <path d="M19,53 C21,84 36,95 60,95 C84,95 99,84 101,53 Z" fill="var(--panel)" stroke="var(--ink)" strokeWidth="2.4" strokeLinejoin="round" />
      <path d="M27,62 C29,80 40,88 60,88" fill="none" stroke="var(--sky)" strokeWidth="2" strokeLinecap="round" opacity=".45" />
      <ellipse cx="45" cy="76" rx="5.5" ry="3.4" fill="var(--blush)" opacity=".7" />
